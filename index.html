function loop() { if (ui.style.display !== 'block') { if (keys['w'] || keys['arrowup']) player.y -= player.speed; if (keys['s'] || keys['arrowdown']) player.y += player.speed; if (keys['a'] || keys['arrowleft']) player.x -= player.speed; if (keys['d'] || keys['arrowright']) player.x += player.speed; if (player.x < 0) player.x = 0; if (player.x > canvas.width - player.size) player.x = canvas.width - player.size; if (player.y < 0) player.y = 0; if (player.y > canvas.height - player.size) player.y = canvas.height - player.size; places.forEach(p => { if (player.x < p.x + p.size && player.x + player.size > p.x && player.y < p.y + p.size && player.y + player.size > p.y) { openQuest(p); } }); } ctx.clearRect(0, 0, canvas.width, canvas.height); // Брусчатка площади ctx.strokeStyle = '#cdbca6'; for(let i=0; i { ctx.fillStyle = p.color; ctx.fillRect(p.x, p.y, p.size, p.size); ctx.strokeStyle = '#2b1a0e'; ctx.lineWidth = 3; ctx.strokeRect(p.x, p.y, p.size, p.size); ctx.fillStyle = '#2b1a0e'; ctx.font = 'bold 14px sans-serif'; ctx.fillText(p.label, p.x + 5, p.y - 10); }); // Персонаж ctx.fillStyle = player.color; ctx.fillRect(player.x, player.y, player.size, player.size); ctx.strokeStyle = '#01579b'; ctx.strokeRect(player.x, player.y, player.size, player.size); requestAnimationFrame(loop); } function openQuest(p) { document.getElementById('q-title').innerText = p.title; document.getElementById('q-text').innerText = p.info; document.getElementById('q-question').innerText = p.question; const ansDiv = document.getElementById('q-answers'); ansDiv.innerHTML = ''; p.answers.forEach((a, i) => { const b = document.createElement('button'); b.className = 'ans-btn'; b.innerText = a; b.onclick = () => { if (i === p.correct) { alert('🎉 Верно!'); closeQuest(); } else { alert('❌ Попробуй еще раз!'); } }; ansDiv.appendChild(b); }); ui.style.display = 'block'; } function closeQuest() { ui.style.display = 'none'; player.x = 400; player.y = 300; } // Инициализация только после полной загрузки окна, чтобы графика не пропадала window.onload = function() { loop(); };
